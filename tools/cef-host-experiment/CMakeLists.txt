cmake_minimum_required(VERSION 3.21)
option(SYNCWATCH_CEF_RESEARCH_BUILD "Compile only under an approved research exception" OFF)
if(NOT SYNCWATCH_CEF_RESEARCH_BUILD)
  message(FATAL_ERROR "Unqualified vendor compilation requires an approved research exception")
endif()
project(SyncWatchCallExperiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
if(APPLE)
  enable_language(OBJCXX)
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/inputs.json" INPUTS)
string(JSON api_version GET "${INPUTS}" apiVersion)
string(JSON EXPECTED_VERSION GET "${INPUTS}" version)
string(JSON INITIAL_NAVIGATION GET "${INPUTS}" initialNavigation)
string(JSON QUALIFIED GET "${INPUTS}" productionQualified)
if(QUALIFIED OR NOT INITIAL_NAVIGATION STREQUAL "about:blank")
  message(FATAL_ERROR "This target is an unqualified blank experiment only")
endif()
set(USE_SANDBOX ON CACHE BOOL "Sandbox is mandatory" FORCE)
set(USE_INSTALLER OFF CACHE BOOL "Use only the explicitly prepared runtime" FORCE)
if(NOT CEF_ROOT OR NOT EXISTS "${CEF_ROOT}/include/cef_version.h")
  message(FATAL_ERROR "Provide the verified experimental SDK through CEF_ROOT")
endif()
file(READ "${CEF_ROOT}/include/cef_version.h" VERSION_HEADER)
string(FIND "${VERSION_HEADER}" "#define CEF_VERSION \"${EXPECTED_VERSION}\"" VERSION_MATCH)
if(VERSION_MATCH EQUAL -1)
  message(FATAL_ERROR "CEF SDK version differs from inputs.json")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)

get_filename_component(PRODUCT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
string(JSON DESIGN_FILE GET "${INPUTS}" designSource)
file(READ "${PRODUCT_ROOT}/${DESIGN_FILE}" GEOMETRY)
foreach(PAIR IN ITEMS "INITIAL_WIDTH:callInitialWidth" "INITIAL_HEIGHT:callInitialHeight"
                      "MINIMUM_WIDTH:callMinimumWidth" "MINIMUM_HEIGHT:callMinimumHeight")
  string(REPLACE ":" ";" PARTS "${PAIR}")
  list(GET PARTS 0 NAME)
  list(GET PARTS 1 TOKEN)
  string(JSON ${NAME} GET "${GEOMETRY}" layout "${TOKEN}" value)
  if(NOT "${${NAME}}" MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "Invalid shared Design dimension")
  endif()
endforeach()
file(READ "${PRODUCT_ROOT}/packages/design/copy.json" COPY)
string(JSON WINDOW_TITLE GET "${COPY}" locales en callWindow)
string(JSON FEATURE_COUNT LENGTH "${INPUTS}" disabledFeatures)
math(EXPR FEATURE_LAST "${FEATURE_COUNT} - 1")
set(DISABLED_FEATURES "")
foreach(INDEX RANGE 0 ${FEATURE_LAST})
  string(JSON FEATURE GET "${INPUTS}" disabledFeatures ${INDEX})
  list(APPEND DISABLED_FEATURES "${FEATURE}")
endforeach()
list(JOIN DISABLED_FEATURES "," DISABLED_FEATURES)
configure_file(src/experiment_config.h.in generated/experiment_config.h @ONLY)

set(CEF_TARGET SyncWatchCallExperiment)
SET_CEF_TARGET_OUT_DIR()
set(COMMON src/experiment.cc src/experiment.h)
if(OS_MAC)
  string(REGEX REPLACE "\\+.*$" "" BUNDLE_VERSION "${EXPECTED_VERSION}")
  set(BUNDLE_EXECUTABLE "${CEF_TARGET}")
  set(BUNDLE_ID "org.salo.syncwatch.call-experiment")
  set(BUNDLE_BACKGROUND false)
  configure_file(macos/Info.plist.in Main-Info.plist @ONLY)
  add_executable(${CEF_TARGET} MACOSX_BUNDLE ${COMMON} src/macos.mm)
  SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
  target_compile_options(${CEF_TARGET} PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  set_target_properties(${CEF_TARGET} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/Main-Info.plist")
  set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
  COPY_MAC_FRAMEWORK(${CEF_TARGET} "${CEF_BINARY_DIR}" "${CEF_APP}")
  foreach(SUFFIXES ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" SUFFIXES "${SUFFIXES}")
    list(GET SUFFIXES 0 NAME_SUFFIX)
    list(GET SUFFIXES 1 TARGET_SUFFIX)
    list(GET SUFFIXES 2 ID_SUFFIX)
    set(HELPER "${CEF_TARGET}_Helper${TARGET_SUFFIX}")
    set(BUNDLE_EXECUTABLE "${CEF_TARGET} Helper${NAME_SUFFIX}")
    set(BUNDLE_ID "org.salo.syncwatch.call-experiment.helper${ID_SUFFIX}")
    set(BUNDLE_BACKGROUND true)
    configure_file(macos/Info.plist.in "${HELPER}-Info.plist" @ONLY)
    add_executable(${HELPER} MACOSX_BUNDLE src/macos_helper.cc)
    SET_EXECUTABLE_TARGET_PROPERTIES(${HELPER})
    set_target_properties(${HELPER} PROPERTIES OUTPUT_NAME "${BUNDLE_EXECUTABLE}"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${HELPER}-Info.plist")
    target_link_libraries(${HELPER} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    add_dependencies(${CEF_TARGET} ${HELPER})
    add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CEF_TARGET_OUT_DIR}/${BUNDLE_EXECUTABLE}.app"
        "${CEF_APP}/Contents/Frameworks/${BUNDLE_EXECUTABLE}.app" VERBATIM)
  endforeach()
elseif(OS_WINDOWS)
  ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_library(${CEF_TARGET} SHARED ${COMMON} src/windows.cc)
  SET_LIBRARY_TARGET_PROPERTIES(${CEF_TARGET})
  COPY_SINGLE_FILE(${CEF_TARGET} "${CEF_BINARY_DIR}/bootstrap.exe" "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.exe")
  target_link_libraries(${CEF_TARGET} libcef_lib)
  SET_LPAC_ACLS(${CEF_TARGET})
elseif(OS_LINUX)
  ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_executable(${CEF_TARGET} ${COMMON} src/linux.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
  target_link_libraries(${CEF_TARGET} libcef_lib)
  set_target_properties(${CEF_TARGET} PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
else()
  message(FATAL_ERROR "Unsupported experiment platform")
endif()

target_include_directories(${CEF_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(${CEF_TARGET} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
if(NOT OS_MAC)
  COPY_FILES(${CEF_TARGET} "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
  COPY_FILES(${CEF_TARGET} "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
  set(NOTICES_DIR "${CEF_TARGET_OUT_DIR}/notices")
else()
  set(NOTICES_DIR "${CEF_APP}/Contents/Resources/notices")
endif()
foreach(NOTICE IN ITEMS LICENSE.txt CREDITS.html README.txt)
  COPY_SINGLE_FILE(${CEF_TARGET} "${CEF_ROOT}/${NOTICE}" "${NOTICES_DIR}/cef/${NOTICE}")
endforeach()
# Keep complete vendor and Chromium notices beside the experimental host.
PRINT_CEF_CONFIG()
